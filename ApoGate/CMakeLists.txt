cmake_minimum_required(VERSION 3.20)
project(DotMicApoGate LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(GATE_QUARTER_GAIN "Temporary fixed 0.25 PCM proof, not production DSP" OFF)
option(GATE_PCM_SWITCH "Temporary per-graph 0.25/no-op proof switch (non-RT read)" OFF)
# This isolated no-op gate deliberately does not link the existing engine, ORT or DSP.
set(GATE_HEADERS "${CMAKE_CURRENT_SOURCE_DIR}/../.deps/apo-win11-headers")
if(NOT EXISTS "${GATE_HEADERS}/audioengineextensionapo.h")
  message(FATAL_ERROR "Run ApoGate/gate.ps1 restore first (pinned Microsoft public SDK headers).")
endif()
add_library(DotMic.ApoGate SHARED noop.cpp exports.def)
# Protected AudioDG must not activate an embedded assembly manifest.
target_link_options(DotMic.ApoGate PRIVATE /MANIFEST:NO)
target_link_libraries(DotMic.ApoGate PRIVATE audiobaseprocessingobject audiomediatypecrt ole32 uuid advapi32)
target_compile_definitions(DotMic.ApoGate PRIVATE _ATL_NO_HOSTING _ATL_NO_PERF_SUPPORT)
if(GATE_QUARTER_GAIN OR GATE_PCM_SWITCH)
  target_compile_definitions(DotMic.ApoGate PRIVATE GATE_QUARTER_GAIN=1)
endif()
if(GATE_PCM_SWITCH)
  target_compile_definitions(DotMic.ApoGate PRIVATE GATE_PCM_SWITCH=1)
endif()
add_executable(DotMic.ApoProbe probe.cpp)
target_link_libraries(DotMic.ApoProbe PRIVATE ole32 uuid propsys setupapi audiomediatypecrt)
foreach(t DotMic.ApoGate DotMic.ApoProbe)
  target_include_directories(${t} BEFORE PRIVATE "${GATE_HEADERS}")
  target_compile_definitions(${t} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_compile_options(${t} PRIVATE /W4 /permissive- /utf-8 /guard:cf)
  target_link_options(${t} PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
endforeach()
# Probe itself does not use ATL; the DLL now uses the official static ATL library.
target_link_options(DotMic.ApoProbe PRIVATE /NODEFAULTLIB:atls.lib)
