cmake_minimum_required(VERSION 3.20)
project(DotMicProductionApo LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(DOTMIC_DEPS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.deps" CACHE PATH "Read-only pinned native dependency root")
set(HEADERS "${DOTMIC_DEPS_ROOT}/apo-win11-headers")
add_library(kissfft STATIC "${DOTMIC_DEPS_ROOT}/kissfft-131.1.0/kiss_fft.c" "${DOTMIC_DEPS_ROOT}/kissfft-131.1.0/kiss_fftr.c")
target_include_directories(kissfft PUBLIC "${DOTMIC_DEPS_ROOT}/kissfft-131.1.0")
set(ORT "${DOTMIC_DEPS_ROOT}/onnxruntime-win-x64-1.23.2")
add_library(DotMic.Inference SHARED inference.cpp)
target_include_directories(DotMic.Inference PRIVATE "${ORT}/include")
target_compile_definitions(DotMic.Inference PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(DotMic.Inference PRIVATE /W4 /permissive- /utf-8 /guard:cf)
target_link_options(DotMic.Inference PRIVATE /MANIFEST:NO /guard:cf /DYNAMICBASE /NXCOMPAT)
target_link_libraries(DotMic.Inference PRIVATE kissfft "${ORT}/lib/onnxruntime.lib" bcrypt)
set(PROOF_DEPENDENCIES "${ORT}/lib/onnxruntime.dll" "${ORT}/lib/onnxruntime_providers_shared.dll")
foreach(name msvcp140 msvcp140_1 msvcp140_2 msvcp140_atomic_wait msvcp140_codecvt_ids vcruntime140 vcruntime140_1 concrt140 vccorlib140)
  list(APPEND PROOF_DEPENDENCIES "${DOTMIC_DEPS_ROOT}/vclibs/${name}.dll")
endforeach()
add_custom_command(TARGET DotMic.Inference POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different ${PROOF_DEPENDENCIES} "$<TARGET_FILE_DIR:DotMic.Inference>"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/../models/dpdfnet2_48khz_hr.onnx" "$<TARGET_FILE_DIR:DotMic.Inference>")
add_library(DotMic.ApoGate SHARED apo.cpp ../ApoGate/exports.def)
target_include_directories(DotMic.ApoGate BEFORE PRIVATE "${HEADERS}")
target_compile_definitions(DotMic.ApoGate PRIVATE _ATL_NO_HOSTING _ATL_NO_PERF_SUPPORT UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(DotMic.ApoGate PRIVATE /W4 /permissive- /utf-8 /guard:cf)
target_link_options(DotMic.ApoGate PRIVATE /MANIFEST:NO /guard:cf /DYNAMICBASE /NXCOMPAT)
target_link_libraries(DotMic.ApoGate PRIVATE audiobaseprocessingobject audiomediatypecrt ole32 uuid propsys rtworkq advapi32)
add_executable(DotMic.ApoCheck tests.cpp)
target_compile_options(DotMic.ApoCheck PRIVATE /W4 /permissive- /utf-8)
add_executable(DotMic.ApoAsyncCheck async-tests.cpp)
target_compile_options(DotMic.ApoAsyncCheck PRIVATE /W4 /permissive- /utf-8)
add_executable(DotMic.ApoMultiMicCheck multimic-tests.cpp)
target_include_directories(DotMic.ApoMultiMicCheck BEFORE PRIVATE "${HEADERS}")
target_compile_definitions(DotMic.ApoMultiMicCheck PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(DotMic.ApoMultiMicCheck PRIVATE /W4 /permissive- /utf-8)
target_link_libraries(DotMic.ApoMultiMicCheck PRIVATE ole32 uuid propsys)
add_executable(DotMic.ApoWorkerCheck worker-tests.cpp)
target_compile_definitions(DotMic.ApoWorkerCheck PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(DotMic.ApoWorkerCheck PRIVATE /W4 /permissive- /utf-8)
target_link_libraries(DotMic.ApoWorkerCheck PRIVATE rtworkq)
add_executable(DotMic.ApoControl cli.cpp)
target_include_directories(DotMic.ApoControl BEFORE PRIVATE "${HEADERS}")
target_compile_definitions(DotMic.ApoControl PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(DotMic.ApoControl PRIVATE /W4 /permissive- /utf-8 /guard:cf)
target_link_options(DotMic.ApoControl PRIVATE /guard:cf /NODEFAULTLIB:atls.lib)
target_link_libraries(DotMic.ApoControl PRIVATE ole32 uuid propsys audiomediatypecrt)
add_library(DotMic.ApoSettings SHARED control.cpp)
target_include_directories(DotMic.ApoSettings BEFORE PRIVATE "${HEADERS}")
target_compile_definitions(DotMic.ApoSettings PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(DotMic.ApoSettings PRIVATE /W4 /permissive- /utf-8 /guard:cf)
target_link_options(DotMic.ApoSettings PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
target_link_libraries(DotMic.ApoSettings PRIVATE ole32 uuid propsys advapi32)
