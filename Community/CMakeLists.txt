cmake_minimum_required(VERSION 3.20)
project(DotMicCommunityIntegration LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
foreach(target DotMic.Integration DotMic.IntegrationProbe)
  if(target STREQUAL "DotMic.Integration")
    add_library(${target} SHARED native.cpp)
  else()
    add_executable(${target} native.cpp)
    target_compile_definitions(${target} PRIVATE DOTMIC_INTEGRATION_TOOL)
  endif()
  target_include_directories(${target} PRIVATE ../.deps/apo-win11-headers)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN)
  target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /guard:cf)
  target_link_options(${target} PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
  target_link_libraries(${target} PRIVATE ole32 uuid propsys audiobaseprocessingobject legacy_stdio_definitions)
endforeach()
add_library(DotMic.CommunitySettings SHARED ../Apo/control.cpp)
set_target_properties(DotMic.CommunitySettings PROPERTIES OUTPUT_NAME "DotMic.ApoSettings")
target_include_directories(DotMic.CommunitySettings PRIVATE ../.deps/apo-win11-headers)
target_compile_definitions(DotMic.CommunitySettings PRIVATE DOTMIC_COMMUNITY_CONTROL UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(DotMic.CommunitySettings PRIVATE /W4 /permissive- /utf-8 /guard:cf)
target_link_options(DotMic.CommunitySettings PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
target_link_libraries(DotMic.CommunitySettings PRIVATE ole32 uuid propsys advapi32)
