cmake_minimum_required(VERSION 3.20)
project(DotMicEntryPoints LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
get_filename_component(ICON "../../App/Assets/DotMic.ico" ABSOLUTE)
configure_file(launcher.rc.in launcher.rc @ONLY)
foreach(target DotMic.AppLauncher DotMic.SetupLauncher)
  add_executable(${target} WIN32 launcher.cpp "${CMAKE_CURRENT_BINARY_DIR}/launcher.rc")
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /guard:cf)
  target_link_options(${target} PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
  target_link_libraries(${target} PRIVATE shell32 user32)
endforeach()
target_compile_definitions(DotMic.SetupLauncher PRIVATE DOTMIC_SETUP_LAUNCHER)
set_target_properties(DotMic.AppLauncher PROPERTIES OUTPUT_NAME "DOT MIC")
set_target_properties(DotMic.SetupLauncher PROPERTIES OUTPUT_NAME "セットアップ")
option(DOTMIC_INSTALLED_SETUP "Shared UI/Setup runtime layout" OFF)
if(DOTMIC_INSTALLED_SETUP)
  target_compile_definitions(DotMic.SetupLauncher PRIVATE DOTMIC_INSTALLED_SETUP)
endif()
add_executable(DotMic.LauncherChecks checks.cpp)
target_compile_definitions(DotMic.LauncherChecks PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(DotMic.LauncherChecks PRIVATE /W4 /permissive- /utf-8)
target_link_libraries(DotMic.LauncherChecks PRIVATE shell32)
